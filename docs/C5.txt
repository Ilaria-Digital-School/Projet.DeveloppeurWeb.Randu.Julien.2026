LDD:

CREATE TABLE Utilisateur(
   utilisateurs VARCHAR(50),
   nom VARCHAR(50) NOT NULL,
   prenom VARCHAR(50),
   email VARCHAR(100),
   mot_de_passe VARCHAR(255) NOT NULL,
   telephone VARCHAR(50),
   interets VARCHAR(50),
   pays VARCHAR(50),
   genre VARCHAR(50),
   role VARCHAR(50) NOT NULL,
   PRIMARY KEY(utilisateurs),
   UNIQUE(email),
   UNIQUE(mot_de_passe)
);

CREATE TABLE Chambre(
   chambre VARCHAR(50),
   hotel VARCHAR(50),
   numero VARCHAR(50),
   type VARCHAR(50),
   capacite INT,
   tarif DECIMAL(15,2),
   description VARCHAR(50),
   PRIMARY KEY(chambre, hotel)
);

CREATE TABLE Reservation(
   reservation VARCHAR(50),
   date_arrivee DATE NOT NULL,
   date_depart DATE NOT NULL,
   nombre_adulte INT NOT NULL,
   nombre_enfant INT NOT NULL,
   montant_total DECIMAL(15,2) NOT NULL,
   statut VARCHAR(50) NOT NULL,
   chambre VARCHAR(50) NOT NULL,
   hotel VARCHAR(50) NOT NULL,
   utilisateurs VARCHAR(50) NOT NULL,
   PRIMARY KEY(reservation),
   FOREIGN KEY(chambre, hotel) REFERENCES Chambre(chambre, hotel),
   FOREIGN KEY(utilisateurs) REFERENCES Utilisateur(utilisateurs)
);

CREATE TABLE Paiement(
   paiement VARCHAR(50),
   montant DECIMAL(15,2) NOT NULL,
   date_paiement DATE,
   statut VARCHAR(50) NOT NULL,
   moyen_paiement VARCHAR(50) NOT NULL,
   reservation VARCHAR(50) NOT NULL,
   PRIMARY KEY(paiement),
   UNIQUE(reservation),
   FOREIGN KEY(reservation) REFERENCES Reservation(reservation)
);

CREATE TABLE Hotel(
   hotel VARCHAR(50),
   nom VARCHAR(50) NOT NULL,
   adresse VARCHAR(50) NOT NULL,
   ville VARCHAR(50),
   code_postale VARCHAR(50),
   pays VARCHAR(50),
   nombre_etoiles INT,
   description VARCHAR(500),
   prestataire VARCHAR(50) NOT NULL,
   chambre VARCHAR(50) NOT NULL,
   hotel_1 VARCHAR(50) NOT NULL,
   PRIMARY KEY(hotel),
   UNIQUE(nom),
   UNIQUE(adresse),
   UNIQUE(prestataire),
   FOREIGN KEY(chambre, hotel_1) REFERENCES Chambre(chambre, hotel)
);

CREATE TABLE Avis(
   avis VARCHAR(50),
   note INT,
   commentaire VARCHAR(50),
   date_avis DATE,
   hotel VARCHAR(50) NOT NULL,
   utilisateurs VARCHAR(50) NOT NULL,
   PRIMARY KEY(avis),
   FOREIGN KEY(hotel) REFERENCES Hotel(hotel),
   FOREIGN KEY(utilisateurs) REFERENCES Utilisateur(utilisateurs)
);

Tables Principales:
